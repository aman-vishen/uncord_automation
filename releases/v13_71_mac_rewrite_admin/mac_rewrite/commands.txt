# app.py controls the reboot/preparation sequence:
# Firmware upgrade
# -> firstboot -y -r
# -> prolinecmd clearall
# -> prolinecmd factorymode set 1
# -> identity write
# -> firstboot -y -r
# -> verify identity after reboot
#
# Supported placeholders:
# {MAC} / {MAC_COLON}
# {MAC_DASH}
# {MAC_NOSEP}
# {SERIAL}
# {GPON}

prolinecmd macaddr set {MAC_NOSEP}
prolinecmd hwsn set {SERIAL}
prolinecmd gponsn set {GPON}
prolinecmd telnetEnable set 1

[VERIFY]
prolinecmd macaddr get
prolinecmd hwsn get
prolinecmd gponsn get

[FINALIZE]
