# ============================================================
# UNCORD ROUTER IDENTITY WRITER COMMANDS
# Source: UNCORD_Router_Command_Template.xlsx (AX3000 / AC1200)
# ============================================================
# Supported placeholders:
# {MAC} / {MAC_COLON} = AA:BB:CC:DD:EE:FF
# {MAC_DASH}          = AA-BB-CC-DD-EE-FF
# {MAC_NOSEP}         = AABBCCDDEEFF
# {SERIAL}            = generated serial number
# {GPON}              = generated GPON number
#
# The workbook states that prolinecmd changes require a reset.
# Therefore the app writes and verifies first, then runs [FINALIZE].

[WRITE]
prolinecmd macaddr set {MAC_NOSEP}
prolinecmd hwsn set {SERIAL}
prolinecmd gponsn set {GPON}
prolinecmd telnetEnable set 1

[VERIFY]
prolinecmd macaddr get
prolinecmd hwsn get
prolinecmd gponsn get

[FINALIZE]
firstboot -y -r
