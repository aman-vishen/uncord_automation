# ============================================================
# UNCORD ROUTER IDENTITY WRITER COMMANDS
# Source: UNCORD_Router_Command_Template.xlsx (AX3000 / AC1200)
# ============================================================
# Supported placeholders:
# {MAC} / {MAC_COLON} = AA:BB:CC:DD:EE:FF
# {MAC_DASH}          = AA-BB-CC-DD-EE-FF
# {MAC_NOSEP}         = AABBCCDDEEFF
# {SERIAL}            = generated serial number
# {GPON}              = generated GPON number
#
# The workbook states that prolinecmd changes require a reset.
# When server firmware update is OFF, the app writes/verifies then runs [FINALIZE].
# When firmware update is ON, [FINALIZE] is skipped because the firmware sysupgrade
# is deliberately the LAST DUT STEP and automatically reboots the PCB.

# ============================================================
# SERVER-CONTROLLED FIRMWARE UPDATE
# ============================================================
# Firmware is not a CLI command in this file. v13.18 runs the supplied
# automatic OpenWrt Python updater directly from mac_writer/firmware_updater.py.
# The central server controls firmware file selection and ON/OFF.
# Identity commands below run FIRST. If they verify successfully and firmware is ON,
# firmware is written LAST; sysupgrade performs the final PCB reboot automatically.

[WRITE]
prolinecmd macaddr set {MAC_NOSEP}
prolinecmd hwsn set {SERIAL}
prolinecmd gponsn set {GPON}
prolinecmd telnetEnable set 1

[VERIFY]
prolinecmd macaddr get
prolinecmd hwsn get
prolinecmd gponsn get

[FINALIZE]
firstboot -y -r
