# ============================================================
# ETE SOLUTIONS INDIA - 8 DUT MAC WRITER CONFIG
# All routers use the SAME DUT IP: 192.168.2.1
# Each DUT is identified by the STATIC IP of its dedicated PC NIC.
#
# Windows 11 NIC setup:
#   DUT1 NIC -> 192.168.2.101 / 255.255.255.0
#   DUT2 NIC -> 192.168.2.102 / 255.255.255.0
#   DUT3 NIC -> 192.168.2.103 / 255.255.255.0
#   DUT4 NIC -> 192.168.2.104 / 255.255.255.0
#   DUT5 NIC -> 192.168.2.105 / 255.255.255.0
#   DUT6 NIC -> 192.168.2.106 / 255.255.255.0
#   DUT7 NIC -> 192.168.2.107 / 255.255.255.0
#   DUT8 NIC -> 192.168.2.108 / 255.255.255.0
#
# Leave DEFAULT GATEWAY blank on these 8 DUT NICs.
# ============================================================

# ============================================================
# LABEL SCAN + AUTOMATIC START
# Operator may scan ONE of: MAC, hardware Serial Number, GPON Serial Number.
# The server resolves the scan to the complete identity row.
# 1 = label scan is mandatory before programming.
# 0 = legacy next-available allocation is allowed.
# ============================================================
REQUIRE_LABEL_SCAN_BEFORE_WRITE = 1
AUTO_START_AFTER_SCAN = 1
PCB_READY_WAIT_SECONDS = 60
PCB_READY_POLL_SECONDS = 0.5

# This command is always sent immediately after Telnet login and before
# any MAC / Serial / GPON write or read command.
PRE_WRITE_COMMAND = prolinecmd clearall

# ============================================================
# SERVER-CONTROLLED FIRMWARE UPDATE / DIRECT PYTHON SSH/SFTP
# ============================================================
# Firmware image + ON/OFF are controlled by the CENTRAL SERVER.
# MAC Writer downloads and SHA-256 verifies the selected image, then uses the
# supplied Python updater flow directly: SSH -> SFTP -> sysupgrade -T ->
# sysupgrade -> automatic reboot -> SSH verify. Firmware is the LAST DUT step.
# No external mcupgrade CLI is required.
# Every SSH socket is bound to ROUTER_n_SOURCE_IP so all 8 DUTs may use 192.168.2.1.
FIRMWARE_SSH_PORT = 22
FIRMWARE_SSH_USERNAME = admin
FIRMWARE_SSH_PASSWORD = admin
FIRMWARE_REMOTE_PATH = /tmp/automatic_firmware.bin

# 0 = preserve OpenWrt configuration (normal sysupgrade).
# 1 = run sysupgrade -n and intentionally erase configuration.
FIRMWARE_ERASE_CONFIG = 0

# Optional post-upgrade text check. Leave blank to only require successful reboot + SSH health check.
FIRMWARE_EXPECTED_VERSION =

FIRMWARE_CONNECT_TIMEOUT_SECONDS = 10
FIRMWARE_VALIDATE_TIMEOUT_SECONDS = 120
FIRMWARE_DOWN_TIMEOUT_SECONDS = 90
FIRMWARE_REBOOT_TIMEOUT_SECONDS = 300
FIRMWARE_POLL_SECONDS = 3

# After the updater verifies SSH, MAC Write waits for production Telnet too.

REQUIRE_SERVER_IDENTIFIERS = 1

# ============================================================
# BOX BUILD / PCB SERIAL GATE
# 1 = Before MAC Write, central server must find PCB Serial Number
#     linked to the identity row by the Box Build station.
#     If PCB Serial Number is blank, this DUT is blocked and nothing is written.
# 0 = Disable this check and allow MAC Write without PCB Serial Number.
# ============================================================
REQUIRE_PCB_SERIAL_BEFORE_WRITE = 1

SERIAL_VALID_REGEX = ^[A-Za-z0-9._/-]{4,64}$
GPON_VALID_REGEX = ^[A-Za-z0-9._/-]{4,64}$
